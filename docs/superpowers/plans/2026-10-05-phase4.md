```python
# src/alfalak/astronomy/LunarCoordinates.py
"""Geocentric low-precision Moon position (Meeus 2nd ed. Ch.47, pp.338-342)."""

import math

from alfalak.astronomy.Astronomical import (
    apparent_obliquity_of_the_ecliptic,
    mean_obliquity_of_the_ecliptic,
)
from alfalak.astronomy.CalendricalHelper import julian_century
from alfalak.util.FloatUtil import unwind_angle

# Transcribe ALL rows of Meeus 2nd ed. Table 47.A (pp.339-340) here.
# Schema: (D, M, M_prime, F, sum_longitude_e6, sum_latitude_e6, sum_radius_m)
# Units as printed: sums in 1e-6 degrees for angles, metres for radius.
# Two example rows shown in exact schema; complete the table from the book,
# then validate with Ex.47.a above. Do not ship a truncated table.
TABLE_47A: list[tuple[int, int, int, int, int, int, int]] = [
    # (D, M, Mp, F, Sl, Sb, Sr) — complete from Meeus pp.339-340
    (0, 0, 1, 0, 6288774, 0, -20905355),
    (2, 0, -1, 0, 1274027, 0, -3699111),
]

class LunarCoordinates:
    """Low-precision geocentric Moon position for a TT Julian Day."""

    def __init__(self, julian_day_tt: float) -> None:
        T = julian_century(julian_day_tt)
        Lp = unwind_angle(218.3164477 + 481267.88123421 * T - 0.0015786 * T * T)
        D = unwind_angle(297.8501921 + 445267.1114034 * T - 0.0018819 * T * T)
        M = unwind_angle(357.5291092 + 35999.0502909 * T - 0.0001536 * T * T)
        Mp = unwind_angle(134.9633964 + 477198.8675055 * T + 0.0087414 * T * T)
        F = unwind_angle(93.2720950 + 483202.0175233 * T - 0.0036539 * T * T)
        sl = sb = sr = 0.0
        for d, m, mp, f, cl, cb, cr in TABLE_47A:
            arg = math.radians(d * D + m * M + mp * Mp + f * F)
            sl += cl * math.sin(arg)
            sb += cb * math.sin(arg)
            sr += cr * math.cos(arg)
        self.longitude: float = unwind_angle(Lp + sl / 1e6)
        self.latitude: float = sb / 1e6
        self.distance_km: float = (385000.56 + sr / 1000.0)
        eps = math.radians(
            apparent_obliquity_of_the_ecliptic(
                T, mean_obliquity_of_the_ecliptic(T)
            )
        )
        lam = math.radians(self.longitude)
        beta = math.radians(self.latitude)
        self.declination: float = math.degrees(
            math.asin(math.sin(beta) * math.cos(eps) + math.cos(beta) * math.sin(eps) * math.sin(lam))
        )
        self.right_ascension: float = unwind_angle(
            math.degrees(
                math.atan2(
                    math.sin(lam) * math.cos(eps) - math.tan(beta) * math.sin(eps),
                    math.cos(lam),
                )
            )
        )
```

```python
# src/alfalak/astronomy/DeltaT.py
"""Delta-T (TT - UTC) provider for the Phase 4 lunar path only."""

import math
import numbers
from decimal import Decimal

from alfalak.exceptions import ValidationError

def delta_t(year: float, override: float | None = None) -> float:
    """Return Delta-T in seconds for a decimal year.

    Polynomial defaults (Espenak): 1986-2005 5th-degree path is folded
    into the 2005-2050 branch below at this precision; 2005-2050:
    ``62.92 + 0.32217*t + 0.005589*t^2`` with ``t = year - 2000``.
    Outside 1986-2050 raise nothing but document historical
    (Morrison/Stephenson) vs modern (prefer IERS observed) in the
    docstring of callers. Prayer path never calls this function.
    """
    if override is not None:
        if isinstance(override, bool) or not isinstance(override, (numbers.Real, Decimal)):
            raise ValidationError(f"Delta-T override must be a real number, got {override!r}.")
        value = float(override)
        if not math.isfinite(value):
            raise ValidationError(f"Delta-T override must be finite, got {override!r}.")
        return value
    if isinstance(year, bool) or not isinstance(year, (numbers.Real, Decimal)):
        raise ValidationError(f"Year must be a real number, got {year!r}.")
    y = float(year)
    if not math.isfinite(y):
        raise ValidationError(f"Year must be finite, got {year!r}.")
    t = y - 2000.0
    return 62.92 + 0.32217 * t + 0.005589 * t * t
```

```python
# src/alfalak/astronomy/CrescentGeometry.py
"""Topocentric Sun-Moon geometry at local sunset (plus best time)."""

from __future__ import annotations

import math
from dataclasses import dataclass
from datetime import date

from alfalak.astronomy.CalendricalHelper import julian_day
from alfalak.astronomy.DeltaT import delta_t
from alfalak.astronomy.LunarCoordinates import LunarCoordinates
from alfalak.astronomy.SolarCoordinates import SolarCoordinates
from alfalak.astronomy.SolarTime import SolarTime
from alfalak.data.Coordinates import Coordinates
from alfalak.util.DateComponents import DateComponents
from alfalak.util.FloatUtil import unwind_angle

@dataclass(frozen=True)
class CrescentGeometry:
    arcl_deg: float
    arcv_geo_deg: float
    arcv_topo_deg: float
    daz_deg: float
    width_arcmin: float
    illumination: float
    lag_hours: float
    moon_age_days: float
    used_delta_t_s: float
    sunset_jd_utc: float

def _altaz(lat: float, lst_deg: float, ra_deg: float, dec_deg: float) -> tuple[float, float]:
    h = math.radians(lst_deg - ra_deg)
    dec = math.radians(dec_deg)
    phi = math.radians(lat)
    alt = math.asin(math.sin(phi) * math.sin(dec) + math.cos(phi) * math.cos(dec) * math.cos(h))
    az = math.atan2(math.sin(h), math.cos(h) * math.sin(phi) - math.tan(dec) * math.cos(phi))
    return math.degrees(alt), unwind_angle(math.degrees(az) + 180.0)

def crescent_geometry_at_sunset(
    day: date,
    coordinates: Coordinates,
    delta_t_override: float | None = None,
) -> CrescentGeometry:
    dc = DateComponents(day.year, day.month, day.day)
    st = SolarTime(dc, coordinates)
    jd_utc = julian_day(day.year, day.month, day.day, 12.0)
    dt = delta_t(day.year + day.timetuple().tm_yday / 365.25, override=delta_t_override)
    jd_tt = jd_utc + dt / 86400.0
    sun = SolarCoordinates(jd_tt)
    moon = LunarCoordinates(jd_tt)
    lst = unwind_angle(sun.apparent_sidereal_time + coordinates.longitude)
    sun_alt, sun_az = _altaz(coordinates.latitude, lst, sun.right_ascension, sun.declination)
    # Parallax correction to topocentric lunar altitude (~1 deg max).
    moon_alt_geo, moon_az = _altaz(coordinates.latitude, lst, moon.right_ascension, moon.declination)
    parallax = math.degrees(math.asin(6378.137 / max(moon.distance_km, 1.0)))
    moon_alt_topo = moon_alt_geo - parallax * math.cos(math.radians(moon_alt_geo))
    arcl = math.degrees(
        math.acos(
            max(-1.0, min(1.0,
                math.sin(math.radians(sun.declination)) * math.sin(math.radians(moon.declination))
                + math.cos(math.radians(sun.declination)) * math.cos(math.radians(moon.declination))
                * math.cos(math.radians(sun.right_ascension - moon.right_ascension))))
        )
    )
    arcv_topo = moon_alt_topo - sun_alt
    arcv_geo = moon_alt_geo - sun_alt
    daz = abs(moon_az - sun_az)
    if daz > 180.0:
        daz = 360.0 - daz
    sd = 0.27245 * parallax * 60.0  # semi-diameter arcmin from parallax
    width = sd * (1.0 - math.cos(math.radians(arcl)))
    illum = (1.0 - math.cos(math.radians(arcl))) / 2.0
    return CrescentGeometry(
        arcl_deg=arcl,
        arcv_geo_deg=arcv_geo,
        arcv_topo_deg=arcv_topo,
        daz_deg=daz,
        width_arcmin=width,
        illumination=illum,
        lag_hours=0.0,
        moon_age_days=1.0,
        used_delta_t_s=dt,
        sunset_jd_utc=jd_utc,
    )
```

```python
"""Yallop (1997) crescent visibility: q-value + zones A-F (NAO TN No.69)."""

import math

def yallop_q(arcv_geo_deg: float, width_arcmin: float) -> float:
    """Best-time q from geocentric ARCV (deg) and topocentric width (arcmin)."""
    w = max(width_arcmin, 1e-6)
    limit = (-0.1018 * w**3 + 0.7319 * w**2 - 6.3226 * w + 11.8371) / 1.0
    return arcv_geo_deg - limit

def yallop_zone(q: float) -> str:
    if q > 0.216:
        return "A"
    if q > -0.014:
        return "B"
    if q > -0.160:
        return "C"
    if q > -0.232:
        return "D"
    if q > -0.293:
        return "E"
    return "F"
```

```python
"""Odeh (2004) crescent visibility: V-value + classes (Exp. Astron. 18:39-64)."""

def odeh_v(arcv_topo_deg: float, width_arcmin: float) -> float:
    """Airless topocentric ARCV (deg) minus limit curve (W in arcmin)."""
    w = max(width_arcmin, 1e-6)
    limit = -0.1018 * w**3 + 0.7319 * w**2 - 6.3226 * w + 7.1651
    return arcv_topo_deg - limit

def odeh_class(v: float, arcl_deg: float) -> str:
    if arcl_deg < 6.4:
        return "D"
    if v >= 5.65:
        return "A"
    if v >= 2.0:
        return "B"
    if v >= -0.96:
        return "C"
    return "D"
```

```python
"""MABIMS crescent presets as thin predicates over 4.3 geometry."""

def is_neo_mabims_2021(alt_deg: float, elong_deg: float) -> bool:
    """KBIR 2016 / adopted 2021: alt >= 3 and elong >= 6.4 at sunset."""
    return alt_deg >= 3.0 and elong_deg >= 6.4

def is_mabims_1992(alt_deg: float, elong_deg: float, age_hours: float) -> bool:
    """Labuan 1992: (alt >= 2 and elong >= 3) or (age >= 8h at moonset)."""
```
